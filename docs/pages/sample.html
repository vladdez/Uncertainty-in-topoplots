make_geography_plot <- function(summary_data, category, title,
                                respondent_total = sum(summary_data$Freq),
                                percent_formatter = \(x) paste0(round(x), "%")) {
  category <- rlang::enquo(category)
  summary_data <- summary_data %>%
    mutate(.percent_label = percent_formatter(percent_score))

  summary_data %>%
    ggplot(aes(percent_score, fct_reorder(!!category, other_last(!!category, percent_score)))) +
    geom_col(width = 0.7, fill = "#6BAED6") +
    geom_text(
      aes(label = .percent_label),
      hjust = -0.15,
      size = 6
    ) +
    scale_x_continuous(
      "Percent of respondents",
      labels = \(x) paste0(x, "%"),
      sec.axis = sec_axis(
        ~ . * respondent_total / 100,
        name = "Number of respondents"
      ),
      expand = expansion(mult = c(0, 0.15))
    ) +
    scale_y_discrete(labels = format_category_labels) +
    labs(title = title, y = NULL) +
    coord_cartesian(clip = "off") +
    theme_classic(base_family = "Lato", base_size = 20) +
    theme(
      plot.title = element_text(hjust = 0.5, face = "bold", size = 22),
      axis.text = element_text(size = 20),
      axis.ticks.y = element_blank(),
      plot.caption.position = "plot",
      plot.caption = element_text(hjust = 0)
    )
}

# Country name followed by its flag (flags line up in a column at the axis).
# Flags: figures/flags/<iso2>.png from flagcdn.com, see figures/flags/SOURCES.md.
flag_tag <- function(iso, height = 11, width = 18) {
  flag_width <- if_else(!is.na(iso) & iso == "ch", height, width)
  sprintf(
    "<img src='%s' width='%d' height='%d'/>",
    project_path("figures", "flags", paste0(iso, ".png")),
    flag_width,
    height
  )
}

flag_label <- function(labels, height = 11, width = 18) {
  labels <- as.character(labels)
  display_labels <- if_else(labels == "United Kingdom", "UK", labels)
  other <- startsWith(labels, "Other")
  iso <- rep(NA_character_, length(labels))
  iso[!other] <- tolower(countrycode::countrycode(
    labels[!other], "country.name", "iso2c",
    custom_match = c(USA = "US", UK = "GB"), warn = FALSE
  ))
  dplyr::case_when(
    labels == "Other" ~ "*Other*",
    labels == "Countries at <2%" ~ "Countries at &lt;2%",
    other ~ sub("^Other \\((.*)\\)$", "*Other*<br>(\\1)", gsub("<", "&lt;", labels)),
    is.na(iso) ~ display_labels,
    TRUE ~ sprintf("%s %s", display_labels, flag_tag(iso, height, width))
  )
}

# Equal Earth projection (Šavrič, Patterson & Jenny, 2018): an equal-area
# pseudocylindrical projection, so country areas are not inflated toward the
# poles as in Mercator.
project_equal_earth <- function(lon, lat) {
  a1 <- 1.340264
  a2 <- -0.081106
  a3 <- 0.000893
  a4 <- 0.003796
  lambda <- lon * pi / 180
  theta <- asin(sqrt(3) / 2 * sin(lat * pi / 180))
  tibble::tibble(
    x = 2 * sqrt(3) * lambda * cos(theta) /
      (3 * (9 * a4 * theta^8 + 7 * a3 * theta^6 + 3 * a2 * theta^2 + a1)),
    y = a4 * theta^9 + a3 * theta^7 + a2 * theta^3 + a1 * theta
  )
}

# Experiment: tool branding in the axis labels. Where a tool has its own
# lettering, only the lettering is used (no icon); otherwise logo + name. Tools
# without a usable logo stay text. Logo sources: figures/software_logos/SOURCES.md
# gridtext places images on the text baseline and cannot lower them, so text
# next to a logo is raised instead: <sup> lifts it by half the font ascent and
# the style attribute restores the full 20 pt size. Logos beside text use
# logo_text_height so their middle lines up with the raised text.
logo_text_height <- 32
logo <- function(file, height = logo_text_height) {
  sprintf(
    "<img src='%s' height='%d'/>",
    project_path("figures", "software_logos", file),
    height
  )
}
# One <sup> lifts about 7 pt; two nested lifts bring the text to the logo middle.
# Branded lettering with descenders (python): one lift aligns the baselines.
lift <- function(text) sprintf("<sup style='font-size:20pt'>%s</sup>", text)
mid <- function(text) {
  sprintf("<sup style='font-size:20pt'><sup style='font-size:20pt'>%s</sup></sup>", text)

make_characteristic_plot <- function(
  plot_data,
  label_column,
  title,
  bar_width = 0.7,
  label_size = 20
) {
  plot_data <- plot_data %>%
    filter(!is.na(.data[[label_column]])) %>%
    mutate(
      display_label = forcats::fct_reorder(
        factor(.data[[label_column]]),
        other_last(.data[[label_column]], percent_score)
      )
    )

  ggplot(plot_data, aes(x = percent_score, y = display_label)) +
    geom_col(width = bar_width, fill = "#6BAED6") +
    geom_text(
      aes(label = paste0(round(percent_score), "%")),
      hjust = -0.15,
      size = 8
    ) +
    scale_x_continuous(
      limits = c(0, 100),
      breaks = seq(0, 100, 25),
      labels = \(x) paste0(x, "%"),
      expand = expansion(mult = c(0, 0.02))
    ) +
    scale_y_discrete(labels = format_category_labels) +
    labs(title = title, x = "Percent of respondents", y = NULL) +
    coord_cartesian(clip = "off") +
    theme_classic(base_family = "Lato", base_size = 20) +
    theme(
      plot.title = element_text(hjust = 0.5, face = "bold", size = 26),
      axis.title.x = element_text(size = 20),
      axis.text.x = element_text(size = 20),
      axis.text.y = element_text(size = label_size),
      axis.ticks.y = element_blank(),
      plot.margin = margin(8, 20, 8, 8)
    )
}

# Parts of a whole as one 100% bar each (Wilke ch. 10); segments ordered as
# the answer scale (level) or by share (field), labelled directly so no legend
# is needed. Segments too narrow for an inside label are labelled to the right.
composition <- bind_rows(
  experience_combined %>%
    transmute(
      question = "Self-assessed level",
      category = level,
      percent_score,
      order = match(level, c("Beginner", "Intermediate", "Expert"))
    ),
  field_combined %>%
    transmute(
      question = "Research field",
      category = field,
      percent_score,
      order = rank(-percent_score, ties.method = "first")
    )
) %>%
  group_by(question) %>%
  arrange(order, .by_group = TRUE) %>%
  mutate(
    xmax = cumsum(percent_score),
    xmin = xmax - percent_score,
    shade = order / max(order),
    inside = percent_score >= 15,
    label = paste0(category, " ", round(percent_score), "%")
  ) %>%
  ungroup() %>%
  mutate(question = factor(question, c("Research field", "Self-assessed level")))

composition_y <- as.numeric(composition$question)

p_composition <- ggplot(composition) +
  geom_rect(
    aes(
      xmin = xmin, xmax = xmax,
      ymin = composition_y - 0.36, ymax = composition_y + 0.36,
      fill = shade
    ),
    colour = "white",
    linewidth = 1
  ) +
  geom_text(
    data = \(d) filter(d, inside),
    aes(
      x = (xmin + xmax) / 2,
      y = as.numeric(question),
      label = label,
      colour = shade > 0.8
    ),
    family = "Lato",
    size = 7,
    lineheight = 0.9

  scale_x_continuous(
    limits = c(0, 100),
    breaks = seq(0, 100, 25),
    labels = \(x) paste0(x, "%"),
