Sample definition

  • Launch date. Records started before 24 Jul 2026 are excluded. They are test runs of the questionnaire: 4 completed records, all from 23 Jul 2026.
  • Completion. Only records that reached page 51 are analysed. Page 51 (open feedback) is the last questionnaire page; pages 52–53 only lead to the optional raffle. Records that stopped there, or that SoSci Survey does not flag as finished, therefore contain the whole questionnaire. 182 records started after launch did not reach page 51.
  • Quality control. 7 participants are flagged for possibly careless answering but kept in all main analyses; see the Quality control page.

# Country name followed by its flag (flags line up in a column at the axis).
# Flags: figures/flags/<iso2>.png from flagcdn.com, see figures/flags/SOURCES.md.
flag_tag <- function(iso, height = 11, width = 18) {
  flag_width <- if_else(!is.na(iso) & iso == "ch", height, width)
  sprintf(
    "<img src='%s' width='%d' height='%d'/>",
    project_path("figures", "flags", paste0(iso, ".png")),
    flag_width,
    height
  )
}

flag_label <- function(labels, height = 11, width = 18) {
  labels <- as.character(labels)
  display_labels <- if_else(labels == "United Kingdom", "UK", labels)
  other <- startsWith(labels, "Other")
  iso <- rep(NA_character_, length(labels))
  iso[!other] <- tolower(countrycode::countrycode(
    labels[!other], "country.name", "iso2c",
    custom_match = c(USA = "US", UK = "GB"), warn = FALSE
  ))
  dplyr::case_when(
    labels == "Other" ~ "*Other*",
    labels == "Countries at <2%" ~ "Countries at &lt;2%",
    other ~ sub("^Other \\((.*)\\)$", "*Other*<br>(\\1)", gsub("<", "&lt;", labels)),
    is.na(iso) ~ display_labels,
    TRUE ~ sprintf("%s %s", display_labels, flag_tag(iso, height, width))
  )
}

# Equal Earth projection (Šavrič, Patterson & Jenny, 2018): an equal-area
# pseudocylindrical projection, so country areas are not inflated toward the
# poles as in Mercator.
project_equal_earth <- function(lon, lat) {
  a1 <- 1.340264
  a2 <- -0.081106
  a3 <- 0.000893
  a4 <- 0.003796
  lambda <- lon * pi / 180
  theta <- asin(sqrt(3) / 2 * sin(lat * pi / 180))
  tibble::tibble(
    x = 2 * sqrt(3) * lambda * cos(theta) /
      (3 * (9 * a4 * theta^8 + 7 * a3 * theta^6 + 3 * a2 * theta^2 + a1)),
    y = a4 * theta^9 + a3 * theta^7 + a2 * theta^3 + a1 * theta
  )
}

# Proportional symbols (Wilke ch. 15; audit V13): counts are shown by circle
# area at each country's centre, so large countries do not dominate the way
# they do in a choropleth of raw counts. The circle sits at the area-weighted
# centroid of the country's largest polygon (its mainland), computed in the
# equal-area projection, so e.g. Alaska or overseas territories do not pull it
# away.
polygon_centroid <- function(x, y) {
  x2 <- c(x[-1], x[1])
  y2 <- c(y[-1], y[1])
  cross <- x * y2 - x2 * y
  signed_area <- sum(cross) / 2  # negative for clockwise polygons
  tibble::tibble(
    cx = sum((x + x2) * cross) / (6 * signed_area),
    cy = sum((y + y2) * cross) / (6 * signed_area),
    area = abs(signed_area)
  )
}

country_symbols <- world_map |>
  filter(participants > 0) |>
  group_by(region, group, participants) |>
  reframe(polygon_centroid(x, y)) |>
  group_by(region) |>
  slice_max(area, n = 1, with_ties = FALSE) |>
  ungroup() |>
  # Large circles first, so smaller ones are drawn on top and stay visible.
  arrange(desc(participants))

  )
}

make_characteristic_plot <- function(
  plot_data,

         labels = print_text$text, value = print_text$small / ggplot2::.pt,
         margin = margin(3, 10, 3, 3))

      hjust = -0.15,
      size = s$value
    ) +
    scale_x_continuous(
      limits = c(0, x_max),
      breaks = seq(0, x_max, if (x_max <= 50) 10 else 25),

      axis.title.x = element_text(size = s$axis),
      axis.text.x = element_text(size = s$axis),
      axis.text.y = element_text(size = s$labels),
      axis.ticks.y = element_blank(),
      plot.margin = s$margin
    )
}

# Print copies of the icon labels (audit V7): showtext draws site text at 96/180
# of its nominal size but print text at true size, while images keep their size
# on both. So font sizes in the markup scale by text_k and image sizes by img_k.
scale_markup <- function(labels, text_k, img_k) {
  scale_number <- function(x, pattern, k, digits) {
    stringr::str_replace_all(x, pattern, \(m) stringr::str_replace(
      m, "[0-9.]+",
      as.character(round(as.numeric(stringr::str_extract(m, "[0-9.]+")) * k, digits))
    ))
  }
  labels %>%
    scale_number("font-size:[0-9.]+pt", text_k, 1) %>%
    scale_number("(width|height)='[0-9.]+'", img_k, 0)
}

# Site version, or with print = TRUE the print version (full journal width,
# 8 pt text, 7 pt ticks and value labels).
# Axis ends: each bar panel gets its own range (no country above 24%, no tool
# above 63%); the panels answer different questions (country sums to 100%,
# software allows several answers), so a shared 0-100% axis only wasted space.
make_participant_characteristics_icons_compact <- function(print = FALSE, icons = TRUE,
                                                         country_max = 40,
                                                         software_max = 75) {
  # Site sizes are nominal (drawn at 96/180 by showtext); print sizes are true pt.
  s <- if (print) {
    list(base = print_text$text, question = print_text$text, ticks = print_text$small,
         segment = print_text$small / ggplot2::.pt, margin = margin(3, 42, 3, 3),
         rect_line = 0.4, tag = print_text$text,
         country = print_text$text, software = print_text$text, lineheight = 0.9)
  } else {
    list(base = 20, question = 22, ticks = 20, segment = 7,
         margin = margin(8, 150, 8, 8), rect_line = 1, tag = 18,
         country = 22, software = 20, lineheight = 0.75)
  }
  # Image sizes follow the text (print text / site text as drawn, 96/180), less
  # 20%: print rows are closer together relative to the text than site rows.
  img_shrink <- 0.8
  country_labels <- \(labels) {
    if (!icons) return(plain_country_labels(labels))
    out <- country_icon_labels(labels)
    if (!print) return(out)
    scale_markup(out, s$country / 22, img_shrink * s$country / (22 * 96 / 180))
  }
  software_labels <- \(labels) {
    if (!icons) return(plain_software_labels(labels))
    out <- software_icon_labels(labels)
    if (!print) return(out)
    scale_markup(out, s$software / 20, img_shrink * s$software / (20 * 96 / 180))
  }

  p_composition <- ggplot(composition) +
    geom_rect(
      aes(
        xmin = xmin, xmax = xmax,
        ymin = composition_y - 0.36, ymax = composition_y + 0.36,
        fill = fill_colour
      ),
      colour = "white",
      linewidth = s$rect_line
    ) +
    geom_text(
      data = \(d) filter(d, inside),
      aes(
        x = (xmin + xmax) / 2,
        y = as.numeric(question),
        label = label,
        colour = shade > 0.8
      ),
      family = "Lato",
      size = s$segment,
      lineheight = 0.9
    ) +
    geom_text(
      data = \(d) filter(d, !inside),
      aes(x = 100, y = as.numeric(question), label = label),
      hjust = -0.08,
      family = "Lato",
      size = s$segment
    ) +
    scale_fill_identity() +
    scale_colour_manual(values = c(`FALSE` = "black", `TRUE` = "white"), guide = "none") +
    scale_x_continuous(
      limits = c(0, 100),
      breaks = seq(0, 100, 25),
      labels = \(x) paste0(x, "%"),
      expand = expansion(mult = c(0, 0))
    ) +
    scale_y_continuous(
      breaks = seq_along(levels(composition$question)),
      labels = levels(composition$question),
      expand = expansion(add = 0.5)
    ) +
    labs(x = "Percent of respondents", y = NULL) +
    coord_cartesian(clip = "off") +
    plot_theme(base_size = s$base) +
    theme(
      axis.text.y = element_text(size = s$question, face = "bold"),
      axis.text.x = element_text(size = s$ticks),
      axis.ticks.y = element_blank(),
      axis.line.y = element_blank(),
      plot.margin = s$margin
    )

  axis_end_mark <- x_axis_end_mark(s$base, size = if (print) 0.55 else 1)

  p_country <- make_characteristic_plot(
    country_over_3, "country_work", "Country of work", print = print,
    x_max = country_max
  ) +
    axis_end_mark +
    scale_y_discrete(labels = country_labels) +
