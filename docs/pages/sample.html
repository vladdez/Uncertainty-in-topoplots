country_responses <- data %>%
  transmute(
    country_work = case_when(
      A004 == "-2" & !is.na(A004s) & A004s != "" ~ A004s,
      TRUE ~ A004
    )
  ) %>%
  mutate(
    country_work = trimws(country_work),
    country_work = case_when(
      country_work %in% c("NL", "Nl", "The Netherlands") ~ "Netherlands",
      country_work == "fr" ~ "France",
      country_work == "UK" ~ "United Kingdom",
      country_work %in% c("United States", "United States of America") ~ "USA",
      country_work %in% c(
        "Prefer not to say",
        "Professor",
        "Senior Researcher"
      ) ~ NA_character_,
      TRUE ~ country_work
    )
  ) %>%
  filter(!is.na(country_work), country_work != "")

make_geography_plot <- function(summary_data, category, title) {
  category <- rlang::enquo(category)
  respondent_total <- sum(summary_data$Freq)

  summary_data %>%
    ggplot(aes(percent_score, fct_reorder(!!category, percent_score))) +
    geom_col(width = 0.7, fill = "#6BAED6") +
    geom_text(
      aes(label = paste0(round(percent_score), "%")),
      hjust = -0.15,
      size = 6
    ) +
    scale_x_continuous(
      "Percent of respondents",
      labels = \(x) paste0(x, "%"),
      sec.axis = sec_axis(
        ~ . * respondent_total / 100,
        name = "Number of respondents"
      ),
      expand = expansion(mult = c(0, 0.15))
    ) +
    labs(title = title, y = NULL) +
    coord_cartesian(clip = "off") +
    theme_classic(base_family = "Lato", base_size = 20) +
    theme(
      plot.title = element_text(hjust = 0.5, face = "bold", size = 22),
      axis.text = element_text(size = 20),
      axis.ticks.y = element_blank(),
      plot.caption.position = "plot",
      plot.caption = element_text(hjust = 0)
    )
}

format_category_labels <- function(labels) {
  lapply(
    labels,
    \(label) {
      if (identical(label, "Other")) {
        bquote(italic(.(label)))
      } else {
        label
      }
    }
  )
}

make_characteristic_plot <- function(
  plot_data,
  label_column,
  title,
  bar_width = 0.7,
  label_size = 20
) {
  plot_data <- plot_data %>%
    filter(!is.na(.data[[label_column]])) %>%
    mutate(
      display_label = forcats::fct_reorder(
        factor(.data[[label_column]]),
        percent_score
      )
    )

  ggplot(plot_data, aes(x = percent_score, y = display_label)) +
    geom_col(width = bar_width, fill = "#6BAED6") +
    geom_text(
      aes(label = paste0(round(percent_score), "%")),
      hjust = -0.15,
      size = 8
    ) +
    scale_x_continuous(
      limits = c(0, 100),
      breaks = seq(0, 100, 25),
      labels = \(x) paste0(x, "%"),
      expand = expansion(mult = c(0, 0.02))
    ) +
    scale_y_discrete(labels = format_category_labels) +
    labs(title = title, x = "Percent of respondents", y = NULL) +
    coord_cartesian(clip = "off") +
    theme_classic(base_family = "Lato", base_size = 20) +
    theme(
      plot.title = element_text(hjust = 0.5, face = "bold", size = 22),
      axis.title.x = element_text(size = 20),
      axis.text.x = element_text(size = 20),
      axis.text.y = element_text(size = label_size),
      axis.ticks.y = element_blank(),
      plot.margin = margin(8, 20, 8, 8)
    )
}
