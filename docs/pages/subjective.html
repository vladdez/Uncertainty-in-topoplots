likert <- data %>%
  select(ID, matches("^A30[1-4]_\\d{2}$")) %>%
  pivot_longer(
    -ID,
    names_to = c("question_code", "number"),
    names_pattern = "(A30[1-4])_(\\d{2})",
    values_to = "rating_raw"
  ) %>%
  mutate(
    number = as.integer(number),

    vistype = factor(
      vistypes[number],
      levels = vistypes
    ),

    question = factor(
      question_code,
      levels = names(question_labels),
      labels = unname(question_labels)
    ),

    rating = as.integer(
      stringr::str_extract(as.character(rating_raw), "^-?\\d+")
    ),

    rating = if_else(
      rating %in% 1:5,
      rating,
      NA_integer_
    )
  ) %>%
  select(ID, question, vistype, rating)

Ordinal mixed-effects models

Code 
likert_model_data <- likert |>
  filter(!is.na(rating)) |>
  mutate(
    ID = factor(ID),
    vistype = droplevels(factor(vistype)),
    rating_ord = ordered(rating, levels = 1:5)
  )

likert_models <- likert_model_data |>
  (\(data) split(data, data$question))() |>
  map(
    \(data_question) clmm(
      rating_ord ~ vistype + (1 | ID),
      data = data_question,
      Hess = TRUE
    )
  )

likert_estimates <- imap_dfr(
  likert_models,
  function(model, question_name) {

    emmeans(
      model,
      ~ vistype,
      mode = "mean.class"
    ) |>
      confint() |>
      as.data.frame() |>
      transmute(
        question = question_name,
        vistype,
        estimated_rating = mean.class,
        lower_ci = asymp.LCL,
        upper_ci = asymp.UCL
      )
  }
)

make_likert_estimate_plot <- function(
  question_name,
  show_x = TRUE,
  show_labels = TRUE
) {

  plot_data <- likert_estimates |>
    filter(question == question_name) |>
    mutate(
      colour_group = if_else(vistype == "basic", "basic", "other"),
    
      display_label = unname(
        likert_display_labels[as.character(vistype)]
      ),
    
      vistype = factor(
        vistype,
        levels = rev(likert_vistype_order)
      )
    )

  p <- ggplot(
    plot_data,
      aes(
    x = estimated_rating,
    y = vistype,
    colour = colour_group
  )
  ) +
    geom_errorbar(
      aes(
        xmin = lower_ci,
        xmax = upper_ci
      ),
      orientation = "y",
      width = 0.12,
      linewidth = 0.55
    ) +
    geom_point(size = 3) +

    # Estimated values
    geom_text(
      aes(label = sprintf("%.1f", estimated_rating)),
      nudge_y = 0.32,
      vjust = 0.5,
      size = 30 / ggplot2::.pt
    ) +

    scale_colour_manual(
  values = c(
    basic = "#3B78B4",
    other = "#222222"
  ),
  guide = "none"
) +

    scale_x_continuous(
      breaks = 1:5,
      labels = function(x) sprintf("%.0f", x),
      expand = expansion(mult = c(0.02, 0.02))
    ) +

    # Hide ordinary y-axis labels
    scale_y_discrete(
      labels = NULL,
      expand = expansion(add = 0.5)
    ) +

    # Show custom labels placed slightly outside the panel
    coord_cartesian(
      xlim = c(1, 5),
      clip = "off"
    ) +

    labs(
      title = unname(likert_panel_titles[question_name]),
      x = NULL,
      y = NULL
    ) +

    theme_classic(base_family = "Lato", base_size = 20) +

    theme(
      plot.title = element_text(
        hjust = 0.5,
        face = "bold",
        size = 50
      ),
      axis.text.x = element_text(size = 20),
      axis.title.x = element_text(size = 20),
      axis.text.y = element_blank(),
      axis.ticks.y = element_blank(),
      axis.line.y = element_blank(),
      panel.background = element_rect(fill = "white", colour = NA),
      plot.background = element_rect(fill = "white", colour = NA),

      # Space for custom labels
       plot.margin = margin(
    t = 5.5,
    r = 5.5,
        b = 8,
    l = if (show_labels) 245 else 12
  )
    )

  if (show_labels) {
    p <- p +
      geom_text(
        data = plot_data,
        aes(
          x = 0.9,
          y = vistype,
          label = display_label,
          colour = colour_group
        ),
        inherit.aes = FALSE,
        hjust = 1,
        size = 14,
        lineheight = 1
      )
  }

  if (!show_x) {
    p <- p +
      theme(
        axis.title.x = element_blank(),
        axis.text.x = element_blank(),
        axis.ticks.x = element_blank()
      )
  }

  p
}
