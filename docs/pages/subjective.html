likert <- data %>%
  select(ID, matches("^A30[1-4]_\\d{2}$")) %>%
  pivot_longer(
    -ID,
    names_to = c("question_code", "number"),
    names_pattern = "(A30[1-4])_(\\d{2})",
    values_to = "rating_raw"
  ) %>%
  mutate(
    number = as.integer(number),

    vistype = factor(
      vistypes[number],
      levels = vistypes
    ),

    question = factor(
      question_code,
      levels = names(question_labels),
      labels = unname(question_labels)
    ),

    rating = as.integer(
      stringr::str_extract(as.character(rating_raw), "^-?\\d+")
    ),

    rating = if_else(
      rating %in% 1:5,
      rating,
      NA_integer_
    )
  ) %>%
  select(ID, question, vistype, rating)

likert_model_data <- likert |>
  filter(!is.na(rating)) |>
  mutate(
    ID = factor(ID),
    vistype = droplevels(factor(vistype)),
    rating_ord = ordered(rating, levels = 1:5)
  )

likert_models <- likert_model_data |>
  (\(data) split(data, data$question))() |>
  map(
    \(data_question) clmm(
      rating_ord ~ vistype + (1 | ID),
      data = data_question,
      Hess = TRUE
    )
  )

likert_estimates <- imap_dfr(
  likert_models,
  function(model, question_name) {

    emmeans(
      model,
      ~ vistype,
      mode = "mean.class"
    ) |>
      confint() |>
      as.data.frame() |>
      transmute(
        question = question_name,
        vistype,
        estimated_rating = mean.class,
        lower_ci = asymp.LCL,
        upper_ci = asymp.UCL
      )
  }
)

question_columns <- unname(question_labels)

make_likert_estimate_plot <- function(question_name, show_x = TRUE) {

  plot_data <- likert_estimates |>
    filter(question == question_name) |>
    arrange(estimated_rating) |>
    mutate(
      colour_group = case_when(
        vistype %in% c("adjacent", "triple_CI") ~ "red",
        vistype == "basic" ~ "blue",
        TRUE ~ "other"
      ),
    
      display_label = unname(
        vistype_labels[as.character(vistype)]
      ),
    
      vistype = factor(
        vistype,
        levels = unique(vistype)
      )
    )

  p <- ggplot(
    plot_data,
      aes(
    x = estimated_rating,
    y = vistype,
    colour = colour_group
  )
  ) +
    geom_errorbar(
      aes(
        xmin = lower_ci,
        xmax = upper_ci
      ),
      orientation = "y",
      width = 0.16,
      linewidth = 0.7
    ) +
    geom_point(size = 3.5) +

    # Estimated values
    geom_text(
      aes(label = sprintf("%.2f", estimated_rating)),
      nudge_y = 0.32,
      vjust = 0.5,
      size = 6
    ) +

    # Custom y-axis labels
    geom_text(
      data = plot_data,
      aes(
        x = 0.78,
        y = vistype,
        label = display_label,
        colour = colour_group
      ),
      inherit.aes = FALSE,
      hjust = 1,
      size = 8,
      lineheight = 0.9
    ) +

    scale_colour_manual(
  values = c(
    red = "#D1495B",
    blue = "#3B78B4",
    other = "#222222"
  ),
  guide = "none"
) +

    scale_x_continuous(
      breaks = 1:5,
      labels = function(x) sprintf("%.0f", x),
      expand = expansion(mult = c(0.02, 0.02))
    ) +

    # Hide ordinary y-axis labels
    scale_y_discrete(
      labels = NULL,
      expand = expansion(add = 0.5)
    ) +

    # Show custom labels placed slightly outside the panel
    coord_cartesian(
      xlim = c(1, 5),
      clip = "off"
    ) +

    labs(
      title = question_name,
      x = "Model-estimated\nexpected rating",
      y = NULL
    ) +

    theme_classic(base_family = "Lato", base_size = 20) +

    theme(
      plot.title = element_text(
        hjust = 0.5,
        face = "bold",
        size = 22
      ),
      axis.text.x = element_text(size = 20),
      axis.title.x = element_text(size = 20),
      axis.text.y = element_blank(),
      axis.ticks.y = element_blank(),

      # Space for custom labels
       plot.margin = margin(
    t = 5.5,
    r = 5.5,
        b = 14,
    l = 200
  )
    )

  if (!show_x) {
    p <- p +
      theme(
        axis.title.x = element_blank(),
        axis.text.x = element_blank(),
        axis.ticks.x = element_blank()
      )
  }

  p
}

feedback_to_pdf <- function(df, title, filename) {

  df <- df %>%
    filter(!is.na(feedback), feedback != "") %>%
    arrange(ID)

  comments <- paste0(
    "<section class='comment'>",
    "<div class='id'>ID: ",
    htmltools::htmlEscape(as.character(df$ID)),
    "</div>",
    "<div class='text'>",
    htmltools::htmlEscape(df$feedback),
    "</div>",
    "</section>"
  )

  html <- paste0(
    "<!DOCTYPE html>
    <html>
    <head>
    <meta charset='UTF-8'>
    <style>
      @page {
        size: A4;
        margin: 18mm;
      }

      body {
        font-family: Arial, sans-serif;
        font-size: 11pt;
        line-height: 1.45;
        color: #222;
      }

      h1 {
        font-size: 18pt;
        margin-bottom: 20px;
      }

      .comment {
        padding: 10px 0 14px 0;
        border-bottom: 1px solid #cccccc;
        page-break-inside: avoid;
      }

      .id {
        font-weight: bold;
        margin-bottom: 5px;
      }

      .text {
        white-space: pre-wrap;
        overflow-wrap: anywhere;
      }
    </style>
    </head>

    <body>
      <h1>", htmltools::htmlEscape(title), "</h1>",
      paste(comments, collapse = "\n"),
    "</body>
    </html>"
  )

  html_file <- tempfile(fileext = ".html")
  writeLines(html, html_file, useBytes = TRUE)

  output_file <- normalizePath(filename, mustWork = FALSE)

  pagedown::chrome_print(
    input = html_file,
    output = output_file
  )

  message("Saved to: ", output_file)
}
