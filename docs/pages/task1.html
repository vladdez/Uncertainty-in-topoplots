# Logistic mixed-effects model
task1_fit <- glmer(
  correct ~ vistype + position_c + (1 | ID),
  data = task1_model,
  family = binomial
)

# Estimated probabilities and 95% confidence intervals
accuracy_estimates <- emmeans(
  task1_fit,
  ~ vistype,
  type = "response"
) |>
  confint() |>
  as.data.frame() |>
  transmute(
    vistype,
    accuracy = prob,
    lower_ci = asymp.LCL,
    upper_ci = asymp.UCL
  ) |>
  arrange(desc(accuracy)) |>
  mutate(
    vistype = factor(vistype, levels = rev(vistype))
  )

Accuracy differences relative to Basic topoplot

This plot shows response-scale differences from Basic topoplot. Confidence intervals crossing zero indicate that the corresponding visualization is not clearly different from Basic at the 95% confidence level.

Code 
task1_response_emmeans <- emmeans(
  task1_fit,
  ~ vistype,
  type = "response",
  regrid = "response"
)

task1_basic_reference <- which(
  task1_response_emmeans@levels$vistype == "basic"
)

task1_basic_contrasts_raw <- contrast(
  task1_response_emmeans,
  method = "trt.vs.ctrl",
  ref = task1_basic_reference
) |>
  confint() |>
  as.data.frame()

task1_lower_ci_column <- grep(
  "lower.CL$|asymp.LCL$",
  names(task1_basic_contrasts_raw),
  value = TRUE
)[1]
task1_upper_ci_column <- grep(
  "upper.CL$|asymp.UCL$",
  names(task1_basic_contrasts_raw),
  value = TRUE
)[1]

task1_basic_contrasts <- task1_basic_contrasts_raw |>
  transmute(
    vistype = stringr::str_remove(contrast, " - basic$"),
    difference = estimate,
    lower_ci = .data[[task1_lower_ci_column]],
    upper_ci = .data[[task1_upper_ci_column]]
  ) |>
  mutate(
    vistype = factor(vistype, levels = levels(accuracy_estimates$vistype))
  )

task1_contrast_limit <- max(
  abs(c(task1_basic_contrasts$lower_ci, task1_basic_contrasts$upper_ci)),
  na.rm = TRUE
)
task1_contrast_limit <- max(
  0.05,
  ceiling(task1_contrast_limit / 0.05) * 0.05
)

task1_contrast_labels <- c(
  adjacent = "Adjacent plots",
  bivariate_corner = "Bivariate corners",
  bivariate_range = "Bivariate range",
  triple_CI = "Confidence intervals",

task1_time_model <- task1_time %>%
  transmute(
    ID = factor(ID),
    vistype = relevel(factor(vistype), ref = "basic"),
    position = as.integer(position),
    time_sec = as.numeric(time_sec)
  ) %>%
  filter(
    !is.na(ID),
    !is.na(vistype),
    !is.na(position),
    !is.na(time_sec),
    time_sec > 0
  ) %>%
  mutate(position_c = position - mean(position))

task1_time_fit <- lmer(
  log(time_sec) ~ vistype + position_c + (1 | ID),
  data = task1_time_model
)

time_estimates <- emmeans(
  task1_time_fit,
  ~ vistype,
  type = "response"
) %>%

  transmute(
    vistype,
    estimated_time = response,
    lower_ci = lower.CL,
    upper_ci = upper.CL
  ) %>%
  arrange(estimated_time) %>%
  mutate(vistype = factor(vistype, levels = vistype))
