# Logistic mixed-effects model
task1_fit <- glmer(
  correct ~ vistype + position_c + (1 | ID),
  data = task1_model,
  family = binomial
)

# Estimated probabilities and 95% confidence intervals
accuracy_estimates <- emmeans(
  task1_fit,
  ~ vistype,
  type = "response"
) |>
  confint() |>
  as.data.frame() |>
  transmute(
    vistype,
    accuracy = prob,
    lower_ci = asymp.LCL,
    upper_ci = asymp.UCL
  ) |>
  arrange(desc(accuracy)) |>
  mutate(
    vistype = factor(vistype, levels = rev(vistype))
  )

accuracy_estimates |>
  mutate(
    vistype = table_vistype(vistype),
    across(
      c(accuracy, lower_ci, upper_ci),
      \(x) scales::percent(x, accuracy = 0.1)
    )
  ) |>
  knitr::kable(
    format = "html",
    col.names = c(
      "Visualization type",
      "Estimated accuracy",
      "Lower 95% CI",
      "Upper 95% CI"
    ),
    caption = "Model-estimated comparison-task accuracy",
    table.attr = kable_table_attributes
  )

Accuracy differences relative to Basic topoplot

This plot shows response-scale differences from Basic topoplot. Confidence intervals crossing zero indicate that the corresponding visualization is not clearly different from Basic at the 95% confidence level.

Code 
task1_response_emmeans <- emmeans(
  task1_fit,
  ~ vistype,
  type = "response",
  regrid = "response"
)

task1_basic_reference <- which(
  task1_response_emmeans@levels$vistype == "basic"
)

task1_basic_contrasts_raw <- contrast(
  task1_response_emmeans,
  method = "trt.vs.ctrl",
  ref = task1_basic_reference
) |>
  confint() |>
  as.data.frame()

task1_lower_ci_column <- grep(
  "lower.CL$|asymp.LCL$",
  names(task1_basic_contrasts_raw),
  value = TRUE
)[1]
task1_upper_ci_column <- grep(
  "upper.CL$|asymp.UCL$",
  names(task1_basic_contrasts_raw),
  value = TRUE
)[1]

task1_basic_contrasts <- task1_basic_contrasts_raw |>
  transmute(
    vistype = stringr::str_remove(contrast, " - basic$"),
    difference = estimate,
    lower_ci = .data[[task1_lower_ci_column]],
    upper_ci = .data[[task1_upper_ci_column]]
  ) |>
  mutate(
    vistype = factor(vistype, levels = levels(accuracy_estimates$vistype))
  )

task1_contrast_limits <- c(
