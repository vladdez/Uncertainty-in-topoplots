score_text <- paste(
  readLines(project_path("regions.txt"), encoding = "UTF-8"),
  collapse = "\n"
)
score_matches <- stringr::str_match_all(
  score_text,
  '"([^"]+)"\\s*=>\\s*array\\(([^)]*)\\)'
)[[1]]
region_scores <- tibble::tibble(
  stimulus = score_matches[, 2],
  values = score_matches[, 3]
) %>%
  tidyr::separate_rows(values, sep = ",\\s*") %>%
  group_by(stimulus) %>%
  mutate(
    region = sprintf("R%02d", row_number()),
    score = as.numeric(values)
  ) %>%
  ungroup() %>%
  select(stimulus, region, score)

Detection-score analysis

The original participant-intercept model is retained as a base sensitivity model. The primary extended model additionally adjusts for spatial pattern and presentation position and includes crossed random intercepts for participants and EEG stimuli.

Code 
model_task2_base <- glmmTMB(
  score ~ vistype + (1 | ID),
  family = ordbeta(link = "logit"),
  data = task2_model
)

model_task2 <- glmmTMB(
  score ~ vistype + pattern + position_c +
    (1 | ID) + (1 | stimulus),
  family = ordbeta(link = "logit"),
  data = task2_model
)

task2_time_model <- task2_time %>%
  transmute(
    ID = factor(ID),
    vistype = relevel(factor(vistype), ref = "basic"),
    position = as.integer(position),
    time_sec = as.numeric(time_sec)
  ) %>%
  filter(complete.cases(.), time_sec > 0) %>%
  mutate(position_c = position - mean(position))

task2_time_fit <- lmer(
  log(time_sec) ~ vistype + position_c + (1 | ID),
  data = task2_time_model
)

task2_time_estimates <- emmeans(
  task2_time_fit,
  ~ vistype,
  type = "response"
) %>%
  confint() %>%

    vistype,
    estimated_time = response,
