  score_text,

  stimulus = score_matches[, 2],
  values = score_matches[, 3]
) %>%

  mutate(

  ) %>%
  ungroup() %>%

Detection-score differences relative to Basic topoplot

This plot shows response-scale differences from Basic topoplot. Confidence intervals crossing zero indicate that the corresponding visualization is not clearly different from Basic at the 95% confidence level.

task2_basic_reference <- which(
  emm_task2_response@levels$vistype == "basic"
)

task2_basic_contrasts_raw <- contrast(
  emm_task2_response,
  method = "trt.vs.ctrl",
  ref = task2_basic_reference
) |>
  confint() |>
  as.data.frame()

task2_lower_ci_column <- grep(
  "lower.CL$|asymp.LCL$",
  names(task2_basic_contrasts_raw),
  value = TRUE
)[1]
task2_upper_ci_column <- grep(
  "upper.CL$|asymp.UCL$",
  names(task2_basic_contrasts_raw),
  value = TRUE
)[1]

task2_basic_contrasts <- task2_basic_contrasts_raw |>
  transmute(
    vistype = stringr::str_remove(contrast, " - basic$"),
    difference = estimate,
    lower_ci = .data[[task2_lower_ci_column]],
    upper_ci = .data[[task2_upper_ci_column]]
  ) |>
  mutate(
    vistype = factor(vistype, levels = levels(task2_estimates$vistype))
  )

task2_contrast_limit <- max(
  abs(c(task2_basic_contrasts$lower_ci, task2_basic_contrasts$upper_ci)),
  na.rm = TRUE
)
task2_contrast_limit <- max(
  0.05,
  ceiling(task2_contrast_limit / 0.05) * 0.05
)

task2_time_model <- task2_time %>%
  transmute(
    ID = factor(ID),
    vistype = relevel(factor(vistype), ref = "basic"),
    position = as.integer(position),
    time_sec = as.numeric(time_sec)
  ) %>%
  filter(complete.cases(.), time_sec > 0) %>%
  mutate(position_c = position - mean(position))

task2_time_fit <- lmer(
  log(time_sec) ~ vistype + position_c + (1 | ID),
  data = task2_time_model
)

task2_time_estimates <- emmeans(
  task2_time_fit,
  ~ vistype,
  type = "response"
) %>%
  confint() %>%
  as.data.frame() %>%
  transmute(
    vistype,
    estimated_time = response,
    lower_ci = lower.CL,
    upper_ci = upper.CL
  ) %>%
  arrange(estimated_time) %>%
  mutate(vistype = factor(vistype, levels = vistype))
